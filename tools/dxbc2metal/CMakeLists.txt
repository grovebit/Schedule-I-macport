cmake_minimum_required(VERSION 3.10)
project(dxbc2metal CXX C)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_OSX_ARCHITECTURES arm64)
set(HLSLCC_LIBRARY_SHARED OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../hlslcc hlslcc_build)
target_compile_definitions(hlslcc PRIVATE _LIBCPP_ENABLE_CXX17_REMOVED_AUTO_PTR=1)
target_compile_options(hlslcc PRIVATE -w)
add_library(dxbc2metal SHARED dxbc2metal.cpp)
set_target_properties(dxbc2metal PROPERTIES CXX_STANDARD 11)
target_compile_definitions(dxbc2metal PRIVATE _LIBCPP_ENABLE_CXX17_REMOVED_AUTO_PTR=1)
target_include_directories(dxbc2metal PRIVATE ../hlslcc/include ../hlslcc/src/internal_includes ../hlslcc/src ../hlslcc/src/cbstring ../hlslcc)
target_link_libraries(dxbc2metal hlslcc)
